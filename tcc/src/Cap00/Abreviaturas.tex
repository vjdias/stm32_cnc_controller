\chapter*{Lista de Abreviaturas}

\begin{description}
  \item[ABN] \quad Canais A, B e N de um encoder incremental.
  \item[API] \quad \emph{Application Programming Interface}.
  \item[APB] \quad \emph{Advanced Peripheral Bus}.
  \item[APP] \quad Camada de aplicação do firmware.
  \item[ARR] \quad \emph{Auto-Reload Register}.
  \item[BLDC] \quad \emph{Brushless Direct Current}.
  \item[CPS] \quad \emph{Counts Per Second}.
  \item[CNC] \quad Controle Numérico Computadorizado.
  \item[CPHA] \quad \emph{Clock Phase}.
  \item[CPOL] \quad \emph{Clock Polarity}.
  \item[CPR] \quad \emph{Counts Per Revolution}.
  \item[CRC] \quad \emph{Cyclic Redundancy Check}.
  \item[CSV] \quad \emph{Comma-Separated Values}.
  \item[DDA] \quad \emph{Digital Differential Analyzer}.
  \item[DMA] \quad \emph{Direct Memory Access}.
  \item[EMC] \quad \emph{Electromagnetic Compatibility}.
  \item[EOF] \quad \emph{End of Frame} (fim do quadro).
  \item[E-STOP] \quad Parada de emergência.
  \item[FPGA] \quad \emph{Field-Programmable Gate Array}.
  \item[FOC] \quad \emph{Field-Oriented Control}.
  \item[FOPDT] \quad \emph{First-Order Plus Dead Time}.
  \item[GUI] \quad \emph{Graphical User Interface}.
  \item[GPIO] \quad \emph{General Purpose Input/Output}.
  \item[HAL] \quad \emph{Hardware Abstraction Layer}.
  \item[IAE] \quad \emph{Integral of Absolute Error}.
  \item[ISR] \quad \emph{Interrupt Service Routine}.
  \item[JSON] \quad \emph{JavaScript Object Notation}.
  \item[LDO] \quad \emph{Low-Dropout Regulator}.
  \item[LPTIM] \quad \emph{Low-Power Timer}.
  \item[NVIC] \quad \emph{Nested Vectored Interrupt Controller}.
  \item[PCB] \quad \emph{Printed Circuit Board}.
  \item[PID] \quad Proporcional-Integral-Derivativo.
  \item[PMSM] \quad \emph{Permanent Magnet Synchronous Motor}.
  \item[PSC] \quad \emph{Prescaler}.
  \item[PWM] \quad \emph{Pulse Width Modulation}.
  \item[RMS] \quad \emph{Root Mean Square}.
  \item[RPS] \quad \emph{Revolutions Per Second}.
  \item[SOF] \quad \emph{Start of Frame} (início do quadro).
  \item[SoC] \quad \emph{System on Chip}.
  \item[SPI] \quad \emph{Serial Peripheral Interface}.
  \item[STEP/DIR/EN] \quad Sinais de passo, direção e habilitação dos drivers de motor de passo.
  \item[SWV] \quad \emph{Serial Wire Viewer}.
  \item[TIM] \quad Temporizador do microcontrolador STM32.
  \item[TTL] \quad \emph{Transistor-Transistor Logic}.
  \item[UART] \quad \emph{Universal Asynchronous Receiver/Transmitter}.
  \item[UFAL] \quad Universidade Federal de Alagoas.
  \item[USART] \quad \emph{Universal Synchronous/Asynchronous Receiver/Transmitter}.
  \item[VCP] \quad \emph{Virtual COM Port}.
\end{description}
